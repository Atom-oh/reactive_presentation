import cf from 'cloudfront';
const kvs = cf.kvs();
async function handler(event) {
  const req = event.request;
  const pin = req.cookies['mr-region'];  // affinity
  const region = pin ? pin.value
    : (await kvs.get('routing', { format: 'json' })).active;
  const ids = region === 'us-west-2'
    ? ['alb-usw2', 'alb-use1'] : ['alb-use1', 'alb-usw2'];
  cf.createRequestOriginGroup({
    originIds: ids.map(id => ({ originId: id })),
    failoverCriteria: { statusCodes: [500, 502, 503, 504] }
  });
  return req;
}

Behavior: 캐시할 것과 안 할 것

동적 API: 캐시 끄고, 함수가 리전을 고른 origin group으로

PathPattern: /api/* TargetOriginId: og-api # origin group (alb-use1, alb-usw2) CachePolicy: Managed-CachingDisabled OriginRequestPolicy: Managed-AllViewerExceptHostHeader AllowedMethods: [GET, HEAD, OPTIONS, PUT, POST, PATCH, DELETE] ViewerProtocolPolicy: redirect-to-https FunctionAssociations: - EventType: viewer-request Function: mr-region-router # KVS 연결, 앞 슬라이드 코드

버전이 붙은 정적 자산: 길게 캐시, S3는 OAC로만

SPA 라우팅: custom error response 대신 함수로 rewrite

PathPattern: "*" # default behavior TargetOriginId: s3-assets CachePolicy: Managed-CachingOptimized FunctionAssociations: - EventType: viewer-request Function: spa-rewrite # 확장자 없는 경로 → /index.html # CustomErrorResponses(403/404 → 200 index.html)는 배포 전체에 적용되어 # /api/*의 404까지 HTML 200으로 바꿔 버림

Origin failover는 GET만 구한다

조건
설정한 상태 코드(예: 500, 502, 503, 504) 또는 연결 실패, timeout. viewer 메서드가 GET, HEAD, OPTIONS일 때만
stateless
health 상태를 기억하지 않아 모든 요청이 primary부터 시도. 장애 중에는 요청마다 지연이 붙음
timeout 튜닝
connection attempts 1~3 (기본 3) × connection timeout 1~10초 (기본 10) → 기본값이면 넘어가기까지 최대 약 30초
쓰기 요청은 앱이 책임
POST /api/orders는 primary 오류가 그대로 사용자에게 갑니다. idempotency key + client retry, 그리고 KVS 활성 리전 전환으로 새 요청을 옮기세요.
